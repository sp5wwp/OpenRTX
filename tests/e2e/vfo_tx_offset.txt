# variants: default smallscreen
# codeplug: channels.rtxc

# VFO shows the TX frequency while transmitting a channel with an RX/TX
# offset. Load 70cm Rptr (idx 1: RX 446.000 / TX 446.600), then PTT: the
# frequency hero switches 446.000 -> 446.600 and reverts on release. Pin a
# strong signal so the RX S-meter saturates identically (see vfo_mem).
rssi 0
wait_stable 5000 3
# ESC toggles VFO -> memory (loads channel 001), UP steps to 002 70cm Rptr.
key ESC
wait_stable
key UP
wait_stable
screenshot_sync tx_offset_rx.bmp
# PTT engages TX: the hero shows the 446.600 TX frequency + orange meter.
ptt
wait_stable
screenshot_sync tx_offset_tx.bmp
# Release PTT: back to the 446.000 RX frequency.
ptt
wait_stable
screenshot_sync tx_offset_rx2.bmp
quit
