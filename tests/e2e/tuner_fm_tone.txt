# variants: default smallscreen

# The FM CTCSS tone in the VFO mode stack follows the direction: a
# decode (RX) tone shows while receiving and hides while transmitting; an
# encode (TX) tone would show only on TX. Enable a decode tone via
# Settings > FM (CTCSS En. -> Decode), return to the tuner screen, then PTT.
rssi 0
wait_stable 5000 3
key ENTER
wait_stable
# Main menu: Banks/Channels/Contacts/GPS/Messages/Settings -> DOWN x5.
key DOWN DOWN DOWN DOWN DOWN ENTER
wait_stable
# Settings: Display/Time&Date/GPS/Radio/M17/FM -> DOWN x5.
key DOWN DOWN DOWN DOWN DOWN ENTER
wait_stable
# FM: CTCSS Tone / CTCSS En. -> DOWN to CTCSS En., edit, UP to "Decode".
key DOWN
wait_stable
key ENTER
wait_stable
key UP
wait_stable
key ENTER
wait_stable
# ESC back to the tuner screen.
key ESC
wait_stable
key ESC
wait_stable
key ESC
wait_stable
screenshot_sync fm_tone_rx.bmp
# PTT: the decode tone hides (no encode tone enabled).
ptt
wait_rtx TX
wait_stable
screenshot_sync fm_tone_tx.bmp
ptt
wait_rtx RX
wait_stable
quit
