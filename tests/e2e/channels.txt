# variants: default smallscreen
# codeplug: channels.rtxc

# Main-menu Channels browser: list the codeplug channels and move selection.
# channels.rtxc holds four channels (2m Calling, 70cm Rptr, M17 CQ, PMR ch1),
# two banks (2m Repeaters, M17 Nets) and three contacts (CQ M17, W1AW, Local
# Net). Pin a strong signal so the S-meter saturates identically across the
# rtx reconfigure a channel load triggers (see vfo_tune).
rssi 0
wait_stable 5000 3
key ENTER
wait_stable
# Channels is main-menu row idx 1 (Banks, Channels, Contacts, ...)
key DOWN ENTER
wait_stable
screenshot_sync channels_list.bmp
# Move the highlight down to "M17 CQ" (idx 2)
key DOWN DOWN
wait_stable
screenshot_sync channels_selected.bmp
# Turning the volume shows the volume bar over the list; once it closes the
# list is still on the same row.
volume 100
sleep 2000
wait_stable
screenshot_sync channels_after_volume.bmp
# Opening and closing the macro menu comes back on the same row too. A tap
# opens and closes it; a hold past the long press would latch it open.
key MONI
wait_stable
screenshot_sync channels_after_macro.bmp
# ENTER loads the channel and pops to the VFO home: M17 CQ, 433.475.
key ENTER
wait_stable
screenshot_sync channels_loaded.bmp
quit
