# variants: default smallscreen

# Macro menu: holding MONI shows the icon-tile grid of quick actions and a
# number key acts on the current channel. MONI is held past the long press,
# which latches the menu open, so another MONI press closes it. Locking the
# keypad (key 9) shows a lock glyph in the top bar of every screen, including
# the tuner screen after the menu closes.
rssi 0
wait_stable 5000 3
keyhold MONI
wait_stable
screenshot_sync macro_open.bmp
# Key 6 cycles power one step (1.0W -> 2.5W) while MONI is held.
key 6
wait_stable
screenshot_sync macro_power.bmp
# Key 9 locks the keypad. The menu stays open after MONI is released, as the
# hold latched it.
key 9
wait_stable
keyrelease
wait_stable
screenshot_sync macro_latched.bmp
# MONI closes it; the tuner screen shows the lock glyph in its top bar.
key MONI
wait_stable
screenshot_sync macro_tuner_locked.bmp
quit
