# variants: default smallscreen

# Macro menu: holding MONI shows the icon-tile grid of quick actions; a number
# key acts on the current channel while held; releasing MONI closes it. The
# overlay uses the shared top bar and is otherwise opaque (no S-meter), so it
# is deterministic on its own. Locking the keypad (key 9) shows a lock glyph in
# the top bar of every screen, including the VFO after the menu closes.
rssi 0
wait_stable 5000 3
keyhold MONI
wait_stable
screenshot_sync macro_open.bmp
# Key 6 cycles power one step (1.0W -> 2.5W) while MONI is held.
key 6
wait_stable
screenshot_sync macro_power.bmp
# Key 9 locks the keypad; on release the VFO shows the lock glyph in its top bar.
key 9
wait_stable
keyrelease
wait_stable
screenshot_sync macro_vfo_locked.bmp
quit
