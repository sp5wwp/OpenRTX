# variants: default smallscreen
# codeplug: channels.rtxc

# M17 destination editor on the VFO. Load the M17 CQ channel (idx 2), then #
# opens the destination editor: the channel-line slot shows "#" and a bracket
# cursor, UP cycles the character and RIGHT extends onto a new one. Pin a
# strong signal so the saturated S-meter is deterministic (see vfo_tune).
rssi 0
wait_stable 5000 3
key ENTER
wait_stable
key DOWN ENTER
wait_stable
key DOWN DOWN
wait_stable
key ENTER
wait_stable
screenshot_sync m17_loaded.bmp
key HASH
wait_stable
screenshot_sync m17_dst_open.bmp
key UP RIGHT UP
wait_stable
screenshot_sync m17_dst_edit.bmp
quit
