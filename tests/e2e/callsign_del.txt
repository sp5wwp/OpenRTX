# variants: default smallscreen

# M17 callsign character-wheel delete: cycle a character to the ⌫ slot, then
# a move key applies the deletion, with no dedicated key.
wait_stable 5000 3
key ENTER
wait_stable
key DOWN DOWN DOWN DOWN DOWN ENTER
wait_stable
# M17 is Settings row idx 4.
key DOWN DOWN DOWN DOWN ENTER
wait_stable
# Edit Callsign ("N0CALL", cursor on the leading 'N').
key ENTER
wait_stable
# Cycle 'N' down 15 steps: N -> ... -> A -> space -> ⌫ (the delete slot).
key DOWN DOWN DOWN DOWN DOWN DOWN DOWN DOWN
wait_stable
key DOWN DOWN DOWN DOWN DOWN DOWN DOWN
wait_stable
screenshot_sync callsign_del_marker.bmp
# A move key (RIGHT) applies the delete, removing the character: "0CALL".
key RIGHT
wait_stable
screenshot_sync callsign_deleted.bmp
quit
